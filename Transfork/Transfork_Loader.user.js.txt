// ==UserScript==
// @name         Transfork Loader
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Stable loader for modular Gandhi Transform Box / Transfork
// @match        *://www.cocrea.world/*
// @grant        GM_xmlhttpRequest
// @connect      raw.githubusercontent.com
// ==/UserScript==

(function () {
    'use strict';

    const BOOT =
        'https://raw.githubusercontent.com/JRonCamay/gandhi/main/Transfork/bootloader.js';

    const BASE =
        BOOT.substring(0, BOOT.lastIndexOf('/') + 1);

    window.__TransforkBaseURL = BASE;

    window.__TransforkLoadText = function (url) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'GET',
                url: url + '?v=' + Date.now(),
                onload(response) {
                    if (response.status >= 200 && response.status < 300) {
                        resolve(response.responseText);
                    } else {
                        reject(new Error('HTTP ' + response.status + ' loading ' + url));
                    }
                },
                onerror() {
                    reject(new Error('Network error loading ' + url));
                }
            });
        });
    };

    window.__TransforkRunCode = function (code, url) {
        try {
            new Function(code + '\n//# sourceURL=' + url)();
        } catch (error) {
            console.error('[Transfork eval error]', url, error);
            alert('Transfork error in ' + url + ': ' + error.message);
            throw error;
        }
    };

    window.__TransforkLoadText(BOOT)
        .then(code => window.__TransforkRunCode(code, BOOT))
        .catch(error => {
            console.error('[Transfork Loader]', error);
            alert('Transfork Loader: ' + error.message);
        });
})();
