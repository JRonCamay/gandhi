[RoutineDef, J_AdjustLowestLvlPoly,
	
	[If, [MVarGet, jMemRemeshAdjusted, 0] == 0 ,
		[Loop, 1,
			[VarSet,slTheLowestVal , [IGet,"ZScript:Lowest Lvl Polys 1=1k"]]
			[MVarSet, jMemLowestLevel, 0, #slTheLowestVal]
			//Adjusting this slider will automatically adjusts the highest poly slider
			[VarSet, slSubDlvl,[IGet,"ZScript:Highest SubD Lvl"]]
			[VarSet, slCtr, 0]
			[VarSet,slLimit, 3000]
			[Loop,(#slSubDlvl - 1),
				[VarSet, slTempCtr,(#slTheLowestVal * 4) ]
				
				[If, #slTempCtr > #slLimit,
					[LoopExit]
				,]
				[VarSet,slTheLowestVal, #slTempCtr]
				[VarInc, slCtr]
			]
			[MVarSet, jMemRemeshAdjusted, 0,1]
			[MVarSet, jMemHighestLevel, 0, #slTheLowestVal ]
			[ISet, "ZScript:Highest Lvl Polys 1=1k", #slTheLowestVal]
			[MVarSet, jMemHighestLevelSubD, 0,(#slCtr + 1)]
			[ISet, "ZScript:Highest SubD Lvl", (#slCtr + 1)]
			[MVarSet, jMemRemeshAdjusted, 0,0]
		]

	,]
