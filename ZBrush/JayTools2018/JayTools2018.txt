[IButton,"ZPlugin:JayTools2018:-- Analysis --","Inspect current and estimated DynaMesh values",

    [Note,[IGetInfo,"Tool:Current Tool"]]

[IButton,"ZPlugin:JayTools2018:Current Resolution","Show the current DynaMesh resolution",
    [Note,[StrMerge,"Current DynaMesh Resolution: ",[IGet,"Tool:Geometry:Resolution"]]]
]

[IButton,"ZPlugin:JayTools2018:Estimated Output","Estimate output and resolution from the current mesh",
    [VarSet,slTargetK,[IGet,"ZPlugin:JayTools2018:Target Points"]]
    [VarSet,slTargetActual,(#slTargetK*1000)]
    [VarSet,slToolInfo,[IGetInfo,"Tool:Current Tool"]]
    [VarSet,slPolysId,[StrFind,"Polys=",#slToolInfo]]
    [VarSet,slHiddenPolysId,[StrFind,"HiddenPolys=",#slToolInfo]]
    [If,(#slPolysId >= 0) && (#slHiddenPolysId > #slPolysId),
        [VarSet,slCurrentPolys,[StrExtract,#slToolInfo,(#slPolysId+6),(#slHiddenPolysId-2)]]
        [If,#slCurrentPolys < 1,[VarSet,slCurrentPolys,1]]
        [VarSet,slCurrentRes,[IGet,"Tool:Geometry:Resolution"]]
        [If,#slCurrentRes < 1,[VarSet,slCurrentRes,64]]
        [VarSet,slEstimatedRes,(#slCurrentRes*((#slTargetActual/#slCurrentPolys)^^0.5))]
        [Note,[StrMerge,"Current: ",#slCurrentPolys,"  Target: ",#slTargetActual,"  Estimated Resolution: ",#slEstimatedRes]]
    ,
        [Note,"Unable to read the active mesh polygon count."]
    ]
]
