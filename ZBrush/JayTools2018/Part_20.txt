			[Loop, 1000,
				[IShow, 1011]

				[IHide, 1011]

					[IUpdate, 1]
					[LoopExit]
				,
					[VarInc, slSubDivCtr]
				]
				
				[VarSet, slLastUndo, #slCurrentUndo]
				[VarInc, slCtr]
				[If, #slCtr == 1000 ,
					[IUpdate, 1]
					[LoopExit]
				,] 
			]/// END OF LOOP
			[IPress, Tool:Geometry:Copy]
			[RoutineCall, J_RestoreUndoPoint]
			[IPress, "Tool:Geometry:Paste Replace"]
			[RoutineCall, J_MemoryOptimize, 0]
			
		]// END OF MAIN LOOP
		[If, #slAutoCollapseFlags > -1 ,
			[RoutineCall, J_RestoreUndoHistoryMax]
			[RoutineCall, J_CanvasClickInsertMesh]
			[RoutineCall, J_RestoreCameraCanvas]
			[RoutineCall, J_SoloRestore]
			[RoutineCall, J_RestoreScrollBarStatus]
			[IShow, 1011]
			
		,]
	
	]// END OF IFREEZE
	
	[If, #slHasError ,
		[NoteBar, ""]
		[Note, #slErrorToDisplay]
	,]
	
	[IShowActions, 1]
	[IUpdate, 1, 1]
