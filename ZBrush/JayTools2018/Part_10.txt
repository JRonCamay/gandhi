[RoutineDef, J_GrowMaskFunc,
	[VarSet, slAdjust, 1]
	[If, #slAdjust ,
		[RoutineCall, J_GrowMask]
		[VarSet, slAdjust, 0]
	,]
	[If, #slAdjust == 0,
		[ISet, "ZScript:GrowMask", 0]
	,]
	
]

[RoutineDef, J_GrowMask,
	[IShowActions, 0]
	[VarSet, slHasError, 0]
	[VArSet, slErrorToDisplay, ""]
	[IFreeze,
		[Loop, 1,
			[RoutineCall, J_CatchValidityError, slHasError, slErrorToDisplay]
			//...continue
			
			
			[VarSet, slGrowVal, [IGet, "ZScript:GrowMask"]]
			[VarSet, slCtr, 0]
			[If, (#slGrowVal > 0) || (#slGrowVal < 0),
				[RoutineCall, J_CatchValidityError, slHasError, slErrorToDisplay]
				[VarSet, slIsPositive, 0]
				[If, #slGrowVal > 0 ,
					[VarSet, slIsPositive, 1]
					[Note, "Growing masks. Please wait..."]
				,]
				[If, #slGrowVal < 1 ,
					[VarSet, slIsPositive, 0]
					[Note, "Shrinking masks. Please wait..."]
				,]
				[IHide, 1011]
				[RoutineCall, J_StoreCameraCanvas]
				[RoutineCall, J_SoloOn]
				[RoutineCall, J_HideCanvas]
				[IUnPress, Transform:Edit]
				[RoutineCall, J_InitiateProgressBar, 4813,4866]
				[RoutineCall, J_ProgressUpdate, 4814, 0,0,0,0]
				[RoutineCall, J_SaveUndoPoint]
				[RoutineCall, J_SetUndoHistoryMax]
				[VarSet, slHiddenPoints, 0]
				[RoutineCall, J_GetPointsInfo, slHiddenPoints, 0]
				[If, #slHiddenPoints > 0 ,
					[IPress, Tool:Visibility:ShowPt]
					[RoutineCall, J_GetPointsInfo, slHiddenPoints, 0]
					[MergeUndo]
				,]
				[VarSet, slHiddenPointsBefore, #slHiddenPoints]
				[IPress, Tool:Visibility:HidePt] // Hide the points
				[VarSet, slHiddenPoints, 0]
				[RoutineCall, J_GetPointsInfo, slHiddenPoints, 0]
				[If, slHiddenPointsBefore == slHiddenPoints,
					[RoutineCall, J_RestoreUndoPoint]
					[RoutineCall, J_MemoryOptimize, 0]
					[VarSet, slHasError, 1]
					[RoutineCall, J_RestoreUndoHistoryMax]
					[IPress, Transform:Edit]
					[RoutineCall, J_RestoreCameraCanvas]
					[RoutineCall, J_SoloRestore]
					[VarSet, slErrorToDisplay, "There is no mask to evaluate. Action cancelled."]
					[LoopExit] 
				,]
				//..continue
				[RoutineCall, J_RestoreUndoPoint]
				[VarSet, slGrowVal, ABS(#slGrowVal)]
				[Loop, #slGrowVal,
					[RoutineCall, J_ProgressUpdate, 4820, #slGrowVal,#slCtr,4820,4859]
					[If, #slIsPositive ,
						[IPress, "Tool:Masking:Mask Adjust Profile"]
						[IPress, "Tool:Masking:Mask Adjust Profile:Reset"]
						[ISet, "Tool:Masking:Blur", 1]
						[ISet, "Tool:Masking:Mask Adjust Profile:Focal Shift", -50]
						[If, #slCtr > 0 ,
							[MergeUndo]
						,]
					,
						[IPress, "Tool:Masking:Mask Adjust Profile"]
						[IPress, "Tool:Masking:Mask Adjust Profile:Reset"]
						[ISet, "Tool:Masking:Blur", 1]
						[ISet, "Tool:Masking:Mask Adjust Profile:Focal Shift", 30]
						[If, #slCtr > 0 ,
							[MergeUndo]
						,]
						
					]
					
					[IPress, "Tool:Masking:Apply"]
					[RoutineCall, J_ProgressUpdate, 4841, #slGrowVal,#slCtr,4820,4859]
					[If, #slIsPositive ,
						[IPress, "Tool:Masking:Mask Adjust Profile:fV"]
						[ISet, "Tool:Masking:Mask Adjust Profile:Focal Shift", 30]
						[ISet, "Tool:Masking:Blur", 0]
						[MergeUndo]
						[IPress, "Tool:Masking:Apply"]
						[MergeUndo]
						[IPress, "Tool:Masking:Apply"]
					,
						
					
					]
					[RoutineCall, J_ProgressUpdate, 4854, #slGrowVal,#slCtr,4820,4859]
					[VarInc, slCtr ]
					[If, #slCtr == #slGrowVal,
						[IUpdate, 1]
						[LoopExit]
					,]
				]
				[RoutineCall, J_ProgressUpdate, 4861, 0, 0, 0, 0]
				[RoutineCall, J_RestoreUndoHistoryMax]
				[IPress, Transform:Edit]
				[RoutineCall, J_RestoreCameraCanvas]
				[RoutineCall, J_SoloRestore]
				[RoutineCall, J_ProgressUpdate, 4866, 0, 0, 0, 0]
				[RoutineCall, J_ProgressEnd]
			,]
		]
	]
	[If, #slHasError ,
		[NOte, #slErrorToDisplay]
	,]
	[IShowActions, 1]
	[IUpdate, 1, 1]

]
