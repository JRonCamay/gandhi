// ==UserScript==
// @name         BlokSearch Loader
// @namespace    http://tampermonkey.net/
// @version      1.1
// @description  Stable loader for modular Gandhi Block Search / BlokSearch
// @match        *://www.cocrea.world/*
// @grant        GM_xmlhttpRequest
// @connect      raw.githubusercontent.com
// ==/UserScript==

(function () {
    'use strict';

    const BOOT =
        'https://raw.githubusercontent.com/JRonCamay/gandhi/main/BlokSearch/bootloader.js';

    const BASE =
        BOOT.substring(0, BOOT.lastIndexOf('/') + 1);

    function loadText(url) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'GET',
                url: url + '?v=' + Date.now(),
                onload(response) {
                    if (response.status >= 200 && response.status < 300) {
                        resolve(response.responseText);
                        return;
                    }

                    reject(new Error('HTTP ' + response.status + ' loading ' + url));
                },
                onerror() {
                    reject(new Error('Network error loading ' + url));
                }
            });
        });
    }

    function runCode(code, url) {
        new Function(
            'BLOKSEARCH_BASE',
            'BLOKSEARCH_LOAD_TEXT',
            'BLOKSEARCH_RUN_CODE',
            code + '\n//# sourceURL=' + url
        )(
            BASE,
            loadText,
            runCode
        );
    }

    loadText(BOOT)
        .then(code => runCode(code, BOOT))
        .catch(error => {
            console.error('[BlokSearch Loader]', error);
            alert('BlokSearch Loader: ' + error.message);
        });
})();
